cmake_minimum_required(VERSION 3.25)
project(CLAWDistribution VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
option(CLAW_BUILD_PLUGIN "Build the SKSE DLL (Windows only)" ON)
option(CLAW_BUILD_TESTS "Build offline core tests (do not execute automatically)" OFF)
set(CLAW_RUNTIME "UNIVERSAL" CACHE STRING "One CommonLib DLL for SE, AE and VR")
set_property(CACHE CLAW_RUNTIME PROPERTY STRINGS UNIVERSAL)
if(NOT CLAW_RUNTIME STREQUAL "UNIVERSAL")
    message(FATAL_ERROR "The current source builds one UNIVERSAL DLL; use a fresh external build tree, preserving older SE/VR builds")
endif()
if(DEFINED CLAW_CONFIGURED_RUNTIME AND NOT CLAW_CONFIGURED_RUNTIME STREQUAL CLAW_RUNTIME)
    message(FATAL_ERROR "Use a separate build directory for each CLAW runtime")
endif()
set(CLAW_CONFIGURED_RUNTIME "${CLAW_RUNTIME}" CACHE INTERNAL "Runtime fixed for this build directory")
set(CLAW_COMMONLIB_DIR "" CACHE PATH "External configured CommonLibSSE-NG checkout")
set(CLAW_SKEE_HEADER "" CACHE FILEPATH "Pinned official external IPluginInterface.h")
set(CLAW_OBODY_HEADER "" CACHE FILEPATH "Optional pinned official OBody 5.2 public API header")
set(CLAW_PHYSICS_MODEL_HEADER "" CACHE FILEPATH "Optional generated external current-vector physics model")
set(CLAW_PHYSICS_FIXTURE_HEADER "" CACHE FILEPATH "Optional generated external numeric physics fixtures")
include(FetchContent)
set(PUGIXML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(pugixml
    URL https://codeload.github.com/zeux/pugixml/zip/refs/tags/v1.15
    URL_HASH SHA256=0dd4dc16e38bbc1c889d3a59660072450127bac7d19415f3cf5d5d6b25c58b35)
FetchContent_MakeAvailable(pugixml)
add_library(claw-preset-core STATIC src/Preset.cpp shared/MatrixKernel.cpp)
target_include_directories(claw-preset-core PUBLIC include shared)
target_link_libraries(claw-preset-core PUBLIC pugixml::pugixml)
add_executable(claw-preset-tool src/Tool.cpp)
target_link_libraries(claw-preset-tool PRIVATE claw-preset-core)
if(MSVC)
    target_compile_options(claw-preset-core PRIVATE /W4 /permissive- /utf-8)
    # Match JavaScript's ordered binary64 operations; no fused multiply-add.
    target_compile_options(claw-preset-core PUBLIC /fp:strict)
endif()
if(CLAW_BUILD_PLUGIN)
    if(NOT WIN32)
        message(FATAL_ERROR "The SKSE runtime requires Windows")
    endif()
    if(NOT EXISTS "${CLAW_COMMONLIB_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "Set CLAW_COMMONLIB_DIR to an external CommonLibSSE-NG checkout")
    endif()
    if(NOT EXISTS "${CLAW_COMMONLIB_DIR}/include/SKSE/Version.h")
        message(FATAL_ERROR "Universal builds require the current CommonLib SDK")
    endif()
    file(READ "${CLAW_COMMONLIB_DIR}/include/SKSE/Version.h" CLAW_SDK_VERSIONS)
    if(NOT CLAW_SDK_VERSIONS MATCHES "RUNTIME_SSE_1_7_104" OR NOT CLAW_SDK_VERSIONS MATCHES "RUNTIME_VR_1_4_15")
        message(FATAL_ERROR "The SDK must support both SE 1.7.104 and VR 1.4.15; the legacy VR SDK is insufficient")
    endif()
    if(NOT EXISTS "${CLAW_SKEE_HEADER}")
        message(FATAL_ERROR "Set CLAW_SKEE_HEADER to the official external IPluginInterface.h")
    endif()
    file(SHA256 "${CLAW_SKEE_HEADER}" CLAW_SKEE_SHA)
    if(NOT CLAW_SKEE_SHA STREQUAL "875410dd22230f76a014a6841004e7d346f5fa1c748cfff695f9f495234da277")
        message(FATAL_ERROR "SKEE header differs from pinned official API revision")
    endif()
    find_package(spdlog CONFIG REQUIRED)
    find_path(RAPIDCSV_INCLUDE_DIRS "rapidcsv.h" REQUIRED)
    # CLAW does not create trampolines; avoid fetching an unused decoder.
    set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
    set(BUILD_TESTS OFF CACHE BOOL "Disable CommonLib tests" FORCE)
    set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
    set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
    set(ENABLE_SKYRIM_VR ON CACHE BOOL "" FORCE)
    add_subdirectory("${CLAW_COMMONLIB_DIR}" commonlib EXCLUDE_FROM_ALL)
    get_filename_component(CLAW_SKEE_INCLUDE_DIR "${CLAW_SKEE_HEADER}" DIRECTORY)
    set(CLAW_PLUGIN_TARGET CLAWDistribution)
    add_library(${CLAW_PLUGIN_TARGET} SHARED src/Plugin.cpp)
    if(MSVC)
        target_compile_options(${CLAW_PLUGIN_TARGET} PRIVATE /bigobj)
    endif()
    if(CLAW_OBODY_HEADER)
        if(NOT EXISTS "${CLAW_OBODY_HEADER}")
            message(FATAL_ERROR "CLAW_OBODY_HEADER must name the official external OBody API.h")
        endif()
        file(SHA256 "${CLAW_OBODY_HEADER}" CLAW_OBODY_HEADER_SHA256)
        if(NOT CLAW_OBODY_HEADER_SHA256 STREQUAL "42a899e88ba53bcc673d2d8e9513e839f0d661ee1e002514fe3551827a98681d")
            message(FATAL_ERROR "OBody API header differs from the pinned 5.2 source interface")
        endif()
        configure_file("${CLAW_OBODY_HEADER}" "${CMAKE_CURRENT_BINARY_DIR}/claw-obody-api.h" COPYONLY)
        target_include_directories(${CLAW_PLUGIN_TARGET} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        target_compile_definitions(${CLAW_PLUGIN_TARGET} PRIVATE CLAW_OBODY_API_HEADER="claw-obody-api.h")
    endif()
    if(CLAW_PHYSICS_MODEL_HEADER)
        if(NOT EXISTS "${CLAW_PHYSICS_MODEL_HEADER}")
            message(FATAL_ERROR "CLAW_PHYSICS_MODEL_HEADER must name the generated external model header")
        endif()
        file(SHA256 "${CLAW_PHYSICS_MODEL_HEADER}" CLAW_PHYSICS_MODEL_SHA256)
        set(CLAW_PHYSICS_MODEL_SHA256 "${CLAW_PHYSICS_MODEL_SHA256}" CACHE INTERNAL "Compiled physics model header digest" FORCE)
        configure_file("${CLAW_PHYSICS_MODEL_HEADER}" "${CMAKE_CURRENT_BINARY_DIR}/claw-physics-fit-model.h" COPYONLY)
        target_include_directories(${CLAW_PLUGIN_TARGET} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        target_compile_definitions(${CLAW_PLUGIN_TARGET} PRIVATE CLAW_PHYSICS_MODEL_HEADER="claw-physics-fit-model.h")
    endif()
    target_include_directories(${CLAW_PLUGIN_TARGET} PRIVATE "${CLAW_SKEE_INCLUDE_DIR}" "${RAPIDCSV_INCLUDE_DIRS}")
    target_link_libraries(${CLAW_PLUGIN_TARGET} PRIVATE claw-preset-core CommonLibSSE::CommonLibSSE)
    target_precompile_headers(${CLAW_PLUGIN_TARGET} PRIVATE <RE/Skyrim.h> <SKSE/SKSE.h>)
    target_compile_definitions(${CLAW_PLUGIN_TARGET} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX CLAW_RUNTIME_UNIVERSAL=1)
    set_target_properties(${CLAW_PLUGIN_TARGET} PROPERTIES OUTPUT_NAME CLAWDistribution)
endif()
if(CLAW_BUILD_TESTS)
    enable_testing()
    if(CLAW_BUILD_PLUGIN)
        add_executable(claw-runtime-tests tests/RuntimeSupportTests.cpp)
        target_include_directories(claw-runtime-tests PRIVATE include)
        target_link_libraries(claw-runtime-tests PRIVATE CommonLibSSE::CommonLibSSE)
        target_precompile_headers(claw-runtime-tests PRIVATE <RE/Skyrim.h> <SKSE/SKSE.h>)
        add_test(NAME claw-runtime-policy COMMAND claw-runtime-tests)
    endif()
    add_executable(claw-preset-tests tests/CoreTests.cpp)
    target_link_libraries(claw-preset-tests PRIVATE claw-preset-core)
    add_test(NAME claw-preset-core COMMAND claw-preset-tests)
    add_executable(claw-matrix-tests tests/MatrixTests.cpp)
    target_link_libraries(claw-matrix-tests PRIVATE claw-preset-core)
    add_test(NAME claw-matrix COMMAND claw-matrix-tests)
    add_executable(claw-physics-observer-tests tests/PhysicsObserverTests.cpp)
    target_include_directories(claw-physics-observer-tests PRIVATE include)
    add_test(NAME claw-physics-observer COMMAND claw-physics-observer-tests)
    add_executable(claw-source-bootstrap-tests tests/SourceBootstrapTests.cpp)
    target_include_directories(claw-source-bootstrap-tests PRIVATE include)
    add_test(NAME claw-source-bootstrap COMMAND claw-source-bootstrap-tests)
    add_executable(claw-physics-fit-tests tests/PhysicsFitTests.cpp)
    target_include_directories(claw-physics-fit-tests PRIVATE include)
    if(CLAW_PHYSICS_MODEL_HEADER AND CLAW_PHYSICS_FIXTURE_HEADER)
        if(NOT EXISTS "${CLAW_PHYSICS_MODEL_HEADER}")
            message(FATAL_ERROR "CLAW_PHYSICS_MODEL_HEADER must name the generated external model header")
        endif()
        if(NOT EXISTS "${CLAW_PHYSICS_FIXTURE_HEADER}")
            message(FATAL_ERROR "CLAW_PHYSICS_FIXTURE_HEADER must name generated numeric fixtures")
        endif()
        # Numeric fixtures must also work without building the SKSE plugin/SDK.
        configure_file("${CLAW_PHYSICS_MODEL_HEADER}" "${CMAKE_CURRENT_BINARY_DIR}/claw-physics-fit-model.h" COPYONLY)
        configure_file("${CLAW_PHYSICS_FIXTURE_HEADER}" "${CMAKE_CURRENT_BINARY_DIR}/claw-physics-fit-fixtures.h" COPYONLY)
        target_include_directories(claw-physics-fit-tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        target_compile_definitions(claw-physics-fit-tests PRIVATE CLAW_PHYSICS_MODEL_HEADER="claw-physics-fit-model.h" CLAW_PHYSICS_FIXTURE_HEADER="claw-physics-fit-fixtures.h")
    endif()
    if(MSVC)
        target_compile_options(claw-physics-fit-tests PRIVATE /UNDEBUG /bigobj)
        # The generated 180-state test fixture initializer exceeds the default
        # Windows EXE main stack. This affects the test executable only.
        target_link_options(claw-physics-fit-tests PRIVATE /STACK:8388608)
    endif()
    add_test(NAME claw-physics-fit COMMAND claw-physics-fit-tests)
endif()
