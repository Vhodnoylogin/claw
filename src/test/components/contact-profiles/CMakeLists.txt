cmake_minimum_required(VERSION 3.21)
project(CLAWContactProfileNumeric LANGUAGES CXX)
option(CLAW_CONTACT_SINGLE_MODEL "Test a standalone emitted profile header" OFF)
if(NOT EXISTS "${CLAW_CONTACT_PROFILE_HEADER}")
  message(FATAL_ERROR "An explicit generated paired model header is required")
endif()
add_executable(claw-contact-profile-numeric profile_numeric.cpp)
target_compile_features(claw-contact-profile-numeric PRIVATE cxx_std_20)
target_include_directories(claw-contact-profile-numeric PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../main/cpp/include")
target_compile_definitions(claw-contact-profile-numeric PRIVATE CLAW_CONTACT_PROFILE_HEADER="${CLAW_CONTACT_PROFILE_HEADER}")
if(CLAW_CONTACT_SINGLE_MODEL)
  target_compile_definitions(claw-contact-profile-numeric PRIVATE CLAW_CONTACT_SINGLE_MODEL=1)
endif()
if(MSVC)
  target_compile_options(claw-contact-profile-numeric PRIVATE /bigobj)
endif()
